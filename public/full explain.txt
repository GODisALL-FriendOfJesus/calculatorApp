JavaScript
import React, { useState } from 'react';
import './App.css';

function App() {
  const [display, setDisplay] = useState('0');
  const [equation, setEquation] = useState('');

  // Handle number and decimal clicks
  const handleNumber = (num) => {
    setDisplay(prev => (prev === '0' ? String(num) : prev + num));
  };

  // Handle operators (+, -, x, ÷)
  const handleOperator = (op) => {
    // Convert visual 'x' and '÷' to JavaScript operators '*' and '/'
    const jsOp = op === 'x' ? '*' : op === '÷' ? '/' : op;
    setEquation(display + ' ' + jsOp + ' ');
    setDisplay('0');
  };

  // Clear everything (AC)
  const handleClear = () => {
    setDisplay('0');
    setEquation('');
  };

  // Calculate result (=)
  const handleCalculate = () => {
    try {
      const fullExpression = equation + display;
      // eslint-disable-next-line no-eval
      const result = eval(fullExpression);
      setDisplay(String(result));
      setEquation('');
    } catch (error) {
      setDisplay('Error');
    }
  };

  return (
    <div className="calculator">
      <div className="screen">{display}</div>
      <div className="keypad">
        <button onClick={handleClear} className="span-2">AC</button>
        <button onClick={() => handleOperator('÷')} className="operator">÷</button>
        <button onClick={() => handleOperator('x')} className="operator">x</button>

        <button onClick={() => handleNumber(7)}>7</button>
        <button onClick={() => handleNumber(8)}>8</button>
        <button onClick={() => handleNumber(9)}>9</button>
        <button onClick={() => handleOperator('-')} className="operator">-</button>

        <button onClick={() => handleNumber(4)}>4</button>
        <button onClick={() => handleNumber(5)}>5</button>
        <button onClick={() => handleNumber(6)}>6</button>
        <button onClick={() => handleOperator('+')} className="operator">+</button>

        <button onClick={() => handleNumber(1)}>1</button>
        <button onClick={() => handleNumber(2)}>2</button>
        <button onClick={() => handleNumber(3)}>3</button>
        <button onClick={handleCalculate} className="equal operator">=</button>

        <button onClick={() => handleNumber(0)} className="span-2">0</button>
        <button onClick={() => handleNumber('.')}>.</button>
      </div>
    </div>
  );
}

export default App;

3. Logic Utility (src/utils/calculate.js)
Handles all math operations cleanly without eval(), supporting chained calculations (e.g., pressing 9 + 9 + 2 = as seen in the video), +/-, and %.

JavaScript
export default function calculate(obj, buttonName) {
  if (buttonName === 'AC') {
    return {
      total: null,
      next: null,
      operation: null,
    };
  }

  // Handle numbers (0-9)
  if (/^[0-9]+$/.test(buttonName)) {
    if (buttonName === '0' && obj.next === '0') {
      return {};
    }
    if (obj.operation) {
      if (obj.next && obj.next !== '0') {
        return { ...obj, next: obj.next + buttonName };
      }
      return { ...obj, next: buttonName };
    }
    if (obj.next && obj.next !== '0') {
      return {
        next: obj.next + buttonName,
        total: null,
      };
    }
    return {
      next: buttonName,
      total: null,
    };
  }

  // Handle decimal point
  if (buttonName === '.') {
    if (obj.next) {
      if (obj.next.includes('.')) {
        return {};
      }
      return { ...obj, next: `${obj.next}.` };
    }
    if (obj.operation) {
      return { ...obj, next: '0.' };
    }
    if (obj.total) {
      if (obj.total.includes('.')) {
        return {};
      }
      return { ...obj, next: `${obj.total}.` };
    }
    return { ...obj, next: '0.' };
  }

  // Handle equals (=)
  if (buttonName === '=') {
    if (obj.next && obj.operation) {
      return {
        total: operate(obj.total, obj.next, obj.operation),
        next: null,
        operation: null,
      };
    }
    return {};
  }

  // Handle +/- (negate)
  if (buttonName === '+/-') {
    if (obj.next) {
      return { ...obj, next: (-1 * parseFloat(obj.next)).toString() };
    }
    if (obj.total) {
      return { ...obj, total: (-1 * parseFloat(obj.total)).toString() };
    }
    return {};
  }

  // Handle % (percentage)
  if (buttonName === '%') {
    if (obj.next) {
      return { ...obj, next: (parseFloat(obj.next) / 100).toString() };
    }
    if (obj.total) {
      return { ...obj, total: (parseFloat(obj.total) / 100).toString() };
    }
    return {};
  }

  // User pressed an operation (+, -, x, ÷)
  // Case 1: Chaining operations (e.g., 9 + 9 + ...)
  if (obj.operation && obj.next) {
    return {
      total: operate(obj.total, obj.next, obj.operation),
      next: null,
      operation: buttonName,
    };
  }

  // Case 2: Changing the operator before typing the second number
  if (obj.operation && !obj.next) {
    return { ...obj, operation: buttonName };
  }

  // Case 3: Pressing an operator right after '='
  if (!obj.next) {
    return { operation: buttonName };
  }

  // Case 4: Standard first operator press
  return {
    total: obj.next,
    next: null,
    operation: buttonName,
  };
}

function operate(numberOne, numberTwo, operation) {
  const one = parseFloat(numberOne || '0');
  const two = parseFloat(numberTwo || '0');

  switch (operation) {
    case '+':
      return (one + two).toString();
    case '-':
      return (one - two).toString();
    case 'x':
      return (one * two).toString();
    case '÷':
      return two === 0 ? 'Error' : (one / two).toString();
    default:
      return '0';
  }
}
